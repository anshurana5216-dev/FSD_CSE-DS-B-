```jsx
import { useEffect, useState } from "react";
import axios from "axios";

function App() {
  const [availableData, setAvailableData] = useState([]);
  const [displayData, setDisplayData] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  // Fetch data from API
  useEffect(() => {
    axios
      .get("https://jsonplaceholder.typicode.com/posts?_limit=6")
      .then((response) => {
        setAvailableData(response.data);
      })
      .catch((err) => {
        console.error("API Error:", err);
        setError("Failed to fetch data. Please try again.");
      })
      .finally(() => {
        setLoading(false);
      });
  }, []);

  // Add data to inner boxes
  const handleAdd = () => {
    if (availableData.length === 0) {
      return;
    }

    const newItem = availableData[0];

    setDisplayData((prevData) => [...prevData, newItem]);

    setAvailableData((prevData) => prevData.slice(1));
  };

  // Delete data from a selected box
  const handleDelete = (id) => {
    setDisplayData((prevData) =>
      prevData.filter((item) => item.id !== id)
    );
  };

  return (
    <div className="flex min-h-screen items-center justify-center bg-slate-100 p-4">

      {/* Main Card */}
      <div className="h-[700px] w-[700px] max-w-full overflow-y-auto rounded-2xl border-2 border-slate-300 bg-white p-5 shadow-xl">

        {/* Heading */}
        <h1 className="mb-5 text-center text-3xl font-bold text-slate-800">
          API Data Cards
        </h1>

        {/* Add Button */}
        <div className="mb-5 flex justify-center">
          <button
            onClick={handleAdd}
            disabled={loading || availableData.length === 0}
            className="rounded-lg bg-green-600 px-6 py-3 font-semibold text-white transition hover:bg-green-700 disabled:cursor-not-allowed disabled:bg-gray-400"
          >
            {loading
              ? "Loading..."
              : availableData.length === 0
                ? "No More Data"
                : "Add Data"}
          </button>
        </div>

        {/* Loading Message */}
        {loading && (
          <p className="text-center text-slate-500">
            Fetching data from API...
          </p>
        )}

        {/* Error Message */}
        {!loading && error && (
          <p className="text-center text-red-600">
            {error}
          </p>
        )}

        {/* Inner Boxes */}
        {!loading && !error && (
          <div className="grid grid-cols-2 justify-items-center gap-4 sm:grid-cols-3">

            {displayData.map((item) => (
              <div
                key={item.id}
                className="flex h-[200px] w-full min-w-0 flex-col overflow-y-auto rounded-xl border-2 border-blue-400 bg-blue-50 p-3 shadow-md"
              >
                {/* Post Title */}
                <h2 className="mb-2 break-words text-sm font-bold text-blue-900">
                  {item.title}
                </h2>

                {/* Post Description */}
                <p className="mb-3 flex-1 text-xs leading-5 text-slate-700">
                  {item.body}
                </p>

                {/* Delete Button */}
                <button
                  onClick={() => handleDelete(item.id)}
                  className="mt-auto rounded-md bg-red-500 px-3 py-2 text-sm font-semibold text-white transition hover:bg-red-600"
                >
                  Delete
                </button>
```
